/** * ============================================================ * Foreign Employee HR Dashboard — Google Apps Script Backend * ============================================================ * วิธีติดตั้ง: ดูคำอธิบายในแชท (ขั้นตอน 1-6) * * Sheet ชื่อ "Employees" ต้องมีหัวตาราง (แถวที่ 1) ตรงตามนี้ทุกตัวอักษร: * * id | title | firstName | middleName | lastName | gender | nationality | birthDate | * wp_issueDate | wp_expiryDate | wp_status | wp_caseStatus | wp_assignedTo | wp_nextAction | wp_notes | * visa_type | visa_issueDate | visa_expiryDate | visa_status | visa_caseStatus | visa_assignedTo | visa_nextAction | visa_notes | * rpt_lastDate | rpt_notes * ============================================================ */ const SHEET_NAME = 'Employees'; const HEADERS = [ 'id','title','firstName','middleName','lastName','gender','nationality','birthDate', 'wp_issueDate','wp_expiryDate','wp_status','wp_caseStatus','wp_assignedTo','wp_nextAction','wp_notes', 'visa_type','visa_issueDate','visa_expiryDate','visa_status','visa_caseStatus','visa_assignedTo','visa_nextAction','visa_notes', 'rpt_lastDate','rpt_notes' ]; // ───────────────────────────────────────────── // Helper: get (or create) the sheet with headers // ───────────────────────────────────────────── function getSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } // ensure header row exists / is correct const firstRow = sheet.getRange(1, 1, 1, HEADERS.length).getValues()[0]; const needsHeader = HEADERS.some((h, i) => firstRow[i] !== h); if (needsHeader) { sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]); } return sheet; } function jsonOutput_(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // ───────────────────────────────────────────── // Convert a flat sheet row <-> nested employee object // ───────────────────────────────────────────── function rowToEmployee_(row) { return { id: row.id, title: row.title, firstName: row.firstName, middleName: row.middleName, lastName: row.lastName, gender: row.gender, nationality: row.nationality, birthDate: formatDate_(row.birthDate), wp: { issueDate: formatDate_(row.wp_issueDate), expiryDate: formatDate_(row.wp_expiryDate), status: row.wp_status, caseStatus: row.wp_caseStatus, assignedTo: row.wp_assignedTo, nextAction: formatDate_(row.wp_nextAction), notes: row.wp_notes }, visa: { type: row.visa_type, issueDate: formatDate_(row.visa_issueDate), expiryDate: formatDate_(row.visa_expiryDate), visaStatus: row.visa_status, caseStatus: row.visa_caseStatus, assignedTo: row.visa_assignedTo, nextAction: formatDate_(row.visa_nextAction), notes: row.visa_notes }, report: { lastDate: formatDate_(row.rpt_lastDate), notes: row.rpt_notes } }; } function employeeToRow_(emp) { return [ emp.id, emp.title, emp.firstName, emp.middleName, emp.lastName, emp.gender, emp.nationality, emp.birthDate, emp.wp.issueDate, emp.wp.expiryDate, emp.wp.status, emp.wp.caseStatus, emp.wp.assignedTo, emp.wp.nextAction, emp.wp.notes, emp.visa.type, emp.visa.issueDate, emp.visa.expiryDate, emp.visa.visaStatus, emp.visa.caseStatus, emp.visa.assignedTo, emp.visa.nextAction, emp.visa.notes, emp.report.lastDate, emp.report.notes ]; } // Google Sheets auto-converts date-looking strings into Date objects. // This converts them back to plain 'YYYY-MM-DD' strings for the frontend. function formatDate_(v) { if (!v) return ''; if (Object.prototype.toString.call(v) === '[object Date]') { const y = v.getFullYear(), m = String(v.getMonth()+1).padStart(2,'0'), d = String(v.getDate()).padStart(2,'0'); return `${y}-${m}-${d}`; } return String(v); } // ───────────────────────────────────────────── // GET → list all employees // ───────────────────────────────────────────── function doGet(e) { const sheet = getSheet_(); const values = sheet.getDataRange().getValues(); if (values.length < 2) return jsonOutput_({ status: 'ok', data: [] }); const headers = values[0]; const rows = values.slice(1); const employees = rows .map(r => { const obj = {}; headers.forEach((h, i) => obj[h] = r[i]); return obj; }) .filter(obj => obj.id) // skip fully empty rows .map(rowToEmployee_); return jsonOutput_({ status: 'ok', data: employees }); } // ───────────────────────────────────────────── // POST → save (add/update) or delete // Body (text/plain, JSON-encoded): // { action: 'save', employee: {...} } // { action: 'delete', id: 'EMP001' } // ───────────────────────────────────────────── function doPost(e) { const lock = LockService.getScriptLock(); lock.waitLock(10000); // prevent two edits colliding at once try { const body = JSON.parse(e.postData.contents); const sheet = getSheet_(); if (body.action === 'save') { saveEmployee_(sheet, body.employee); return jsonOutput_({ status: 'ok' }); } if (body.action === 'delete') { deleteEmployee_(sheet, body.id); return jsonOutput_({ status: 'ok' }); } return jsonOutput_({ status: 'error', message: 'Unknown action: ' + body.action }); } catch (err) { return jsonOutput_({ status: 'error', message: err.toString() }); } finally { lock.releaseLock(); } } function findRowIndexById_(sheet, id) { const ids = sheet.getRange(2, 1, Math.max(sheet.getLastRow()-1,0), 1).getValues(); for (let i = 0; i < ids.length; i++) { if (ids[i][0] === id) return i + 2; // +2: 1-indexed + header row } return -1; } function saveEmployee_(sheet, employee) { const rowValues = employeeToRow_(employee); const rowIdx = findRowIndexById_(sheet, employee.id); if (rowIdx > 0) { sheet.getRange(rowIdx, 1, 1, rowValues.length).setValues([rowValues]); } else { sheet.appendRow(rowValues); } } function deleteEmployee_(sheet, id) { const rowIdx = findRowIndexById_(sheet, id); if (rowIdx > 0) sheet.deleteRow(rowIdx); }
